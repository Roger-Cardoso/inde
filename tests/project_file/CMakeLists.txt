cmake_minimum_required(VERSION 3.22)
project(inde_p5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3>=3.37)
option(P5_SANITIZE "Instrument the P5 store, parser and command-line adapter" OFF)
if(P5_SANITIZE)
  add_compile_options(-g1 -fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
get_filename_component(P5_ROOT ../.. ABSOLUTE)
execute_process(COMMAND ${Python3_EXECUTABLE} ${P5_ROOT}/verificar_base.py --integrado COMMAND_ERROR_IS_FATAL ANY)
set(P5_BASE ${P5_ROOT}/fontes/inde-integrado-schema37)
add_library(inde_p5_owner_json STATIC ${P5_BASE}/src/persistence/json.cpp)
target_include_directories(inde_p5_owner_json PUBLIC ${P5_BASE}/include)
# The pinned owner's warnings match its native build. Strict conversion
# diagnostics apply to every new P5 unit without modifying frozen source.
target_compile_options(inde_p5_owner_json PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(inde_project_file STATIC ${P5_ROOT}/cpp/project_file.cpp ${P5_ROOT}/cpp/project_file_transport.cpp ${P5_ROOT}/cpp/project_file_session.cpp)
target_include_directories(inde_project_file PUBLIC ${P5_ROOT}/cpp ${P5_BASE}/include)
target_link_libraries(inde_project_file PUBLIC inde_p5_owner_json PkgConfig::GLIB PkgConfig::SQLITE)
add_executable(inde_project_file_tests ${P5_ROOT}/cpp/project_file_tests.cpp)
target_link_libraries(inde_project_file_tests PRIVATE inde_project_file)
add_test(NAME inde_project_file_tests COMMAND inde_project_file_tests)
set_tests_properties(inde_project_file_tests PROPERTIES TIMEOUT 120)

add_executable(inde_project_file_cli ${P5_ROOT}/cpp/project_file_main.cpp)
set_target_properties(inde_project_file_cli PROPERTIES OUTPUT_NAME inde_project_file)
target_link_libraries(inde_project_file_cli PRIVATE inde_project_file)
foreach(p5_target inde_project_file inde_project_file_tests inde_project_file_cli)
  target_compile_options(${p5_target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endforeach()
add_test(NAME inde_project_file_transport_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_transport.py $<TARGET_FILE:inde_project_file_cli>)
set_tests_properties(inde_project_file_transport_tests PROPERTIES TIMEOUT 180)

add_executable(inde_project_file_session_tests ${P5_ROOT}/cpp/project_file_session_tests.cpp)
target_link_libraries(inde_project_file_session_tests PRIVATE inde_project_file)
target_compile_options(inde_project_file_session_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
add_test(NAME inde_project_file_session_tests COMMAND inde_project_file_session_tests)
set_tests_properties(inde_project_file_session_tests PROPERTIES TIMEOUT 180)

add_executable(inde_project_file_benchmark ${P5_ROOT}/cpp/project_file_benchmark.cpp)
target_compile_definitions(inde_project_file_benchmark PRIVATE P5C_BLOCK_BENCHMARK=1)
target_link_libraries(inde_project_file_benchmark PRIVATE inde_project_file)
target_compile_options(inde_project_file_benchmark PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
if(BUILD_TESTING)
  add_test(NAME inde_project_file_block_session_tests COMMAND ${CMAKE_COMMAND} -E env P5_TRANSPORT_BLOCKS=1 $<TARGET_FILE:inde_project_file_session_tests>)
  set_tests_properties(inde_project_file_block_session_tests PROPERTIES TIMEOUT 180)
endif()
