cmake_minimum_required(VERSION 3.22)
project(inde_obj01_foundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OBJ01_SANITIZE "Use ASan and UBSan" OFF)
include(CTest)
set(OBJ01_CPP ${CMAKE_CURRENT_LIST_DIR}/../../cpp)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Werror)
if(OBJ01_SANITIZE)
  add_compile_options(-g -fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(inde_object_model STATIC ${OBJ01_CPP}/object_model.cpp ${OBJ01_CPP}/object_commands.cpp)
target_include_directories(inde_object_model PUBLIC ${OBJ01_CPP})
foreach(suite object_model object_commands)
  add_executable(inde_${suite}_tests ${OBJ01_CPP}/${suite}_tests.cpp)
  target_link_libraries(inde_${suite}_tests PRIVATE inde_object_model)
  add_test(NAME inde_${suite}_tests COMMAND inde_${suite}_tests)
  set_tests_properties(inde_${suite}_tests PROPERTIES TIMEOUT 120)
endforeach()
