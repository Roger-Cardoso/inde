cmake_minimum_required(VERSION 3.22)
project(inde_obj01_composite_p2a LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(PkgConfig REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../object_document document-domain)
pkg_check_modules(PILOT_PIXBUF REQUIRED IMPORTED_TARGET gdk-pixbuf-2.0)
pkg_check_modules(PILOT_SQLITE REQUIRED IMPORTED_TARGET sqlite3)
find_package(ZLIB REQUIRED)
add_library(inde_object_composite_pilot_domain STATIC ../../cpp/object_composite_pilot.cpp)
target_link_libraries(inde_object_composite_pilot_domain PUBLIC inde_object_document_adapter PRIVATE PkgConfig::PILOT_PIXBUF PkgConfig::PILOT_SQLITE ZLIB::ZLIB)
target_compile_features(inde_object_composite_pilot_domain PUBLIC cxx_std_20)
target_compile_options(inde_object_composite_pilot_domain PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
add_executable(inde_object_composite_pilot ../../cpp/object_composite_pilot_main.cpp)
target_link_libraries(inde_object_composite_pilot PRIVATE inde_object_composite_pilot_domain)
target_compile_options(inde_object_composite_pilot PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
if(OBJ01_SANITIZE)
  foreach(t inde_object_composite_pilot_domain inde_object_composite_pilot)
    target_compile_options(${t} PRIVATE -g1 -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
add_test(NAME inde_object_composite_pilot_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_pilot.py $<TARGET_FILE:inde_object_composite_pilot>)
set_tests_properties(inde_object_composite_pilot_tests PROPERTIES TIMEOUT 180)
add_test(NAME inde_object_composite_http_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_http.py $<TARGET_FILE:inde_object_composite_pilot>)
set_tests_properties(inde_object_composite_http_tests PROPERTIES TIMEOUT 180)

add_test(NAME inde_object_composite_p2b_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_p2b.py $<TARGET_FILE:inde_object_composite_pilot>)
set_tests_properties(inde_object_composite_p2b_tests PROPERTIES TIMEOUT 180)
add_test(NAME inde_object_composite_http_p2b_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_http_p2b.py $<TARGET_FILE:inde_object_composite_pilot>)
set_tests_properties(inde_object_composite_http_p2b_tests PROPERTIES TIMEOUT 180)
# Fault injection exists only in this separate test driver, never in the UI binary.
add_executable(inde_object_composite_pilot_fault_driver ../../cpp/object_composite_pilot_main.cpp)
target_compile_definitions(inde_object_composite_pilot_fault_driver PRIVATE OBJ01_TEST_FAULTS=1)
target_link_libraries(inde_object_composite_pilot_fault_driver PRIVATE inde_object_composite_pilot_domain)
target_compile_options(inde_object_composite_pilot_fault_driver PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
if(OBJ01_SANITIZE)
  target_compile_options(inde_object_composite_pilot_fault_driver PRIVATE -g1 -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(inde_object_composite_pilot_fault_driver PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME inde_object_composite_p2c_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_p2c.py $<TARGET_FILE:inde_object_composite_pilot> $<TARGET_FILE:inde_object_composite_pilot_fault_driver>)
set_tests_properties(inde_object_composite_p2c_tests PROPERTIES TIMEOUT 300)

add_test(NAME inde_object_composite_http_p2c_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_http_p2c.py $<TARGET_FILE:inde_object_composite_pilot>)
set_tests_properties(inde_object_composite_http_p2c_tests PROPERTIES TIMEOUT 180)
