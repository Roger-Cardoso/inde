cmake_minimum_required(VERSION 3.22)
project(inde_web_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(cmake/Optimization.cmake)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
pkg_check_modules(PIXBUF REQUIRED IMPORTED_TARGET gdk-pixbuf-2.0)
pkg_check_modules(GTKMM REQUIRED IMPORTED_TARGET gtkmm-4.0>=4.8)
set(BUILD_TESTING ${INDE_BUILD_CORE_TESTS} CACHE BOOL "Core regression tests" FORCE)
include(CTest)
set(INDE_FROZEN_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../fontes/inde-integrado-schema37)
set(INDE_ACTIVE_SOURCE ${CMAKE_BINARY_DIR}/inde-source)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  ${CMAKE_CURRENT_SOURCE_DIR}/base_patches.json ${CMAKE_CURRENT_SOURCE_DIR}/otimizar_base.py)
set(INDE_OPTIMIZATION_PHASE OFF)
if(INDE_CODE_OPTIMIZATIONS)
  set(INDE_OPTIMIZATION_PHASE ON)
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/otimizar_base.py
  ${INDE_FROZEN_SOURCE} ${INDE_ACTIVE_SOURCE} ${INDE_OPTIMIZATION_PHASE}
  COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory(${INDE_ACTIVE_SOURCE} inde-base EXCLUDE_FROM_ALL)
add_library(inde_web_publication snapshot_model.cpp publication_service.cpp public_identity.cpp comic_model.cpp comic_service.cpp)
target_include_directories(inde_web_publication PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(inde_web_publication PUBLIC inde_production PkgConfig::GLIB PkgConfig::PIXBUF)
target_compile_options(inde_web_publication PRIVATE -Wall -Wextra -Wpedantic)
add_executable(inde_web_bridge snapshot.cpp catalog_demo.cpp import_literature.cpp import_work.cpp)
target_link_libraries(inde_web_bridge PRIVATE inde_web_publication)
target_compile_options(inde_web_bridge PRIVATE -Wall -Wextra -Wpedantic)

find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/integrar_gtk.py
  ${INDE_ACTIVE_SOURCE} ${CMAKE_CURRENT_SOURCE_DIR}/overlay COMMAND_ERROR_IS_FATAL ANY)
get_target_property(INDE_UI_SOURCES inde SOURCES)
list(REMOVE_ITEM INDE_UI_SOURCES src/main.cpp src/ui/main_window.cpp src/ui/overlay_dialog.cpp src/ui/workspace_shell.cpp src/ui/main_window_context.cpp src/ui/main_window_keyboard.cpp)
set_property(TARGET inde PROPERTY SOURCES "${INDE_UI_SOURCES}")
target_sources(inde PRIVATE overlay/src/main.cpp overlay/src/ui/main_window.cpp overlay/src/ui/overlay_dialog.cpp
  workspace_shell.cpp overlay/src/ui/main_window_context.cpp overlay/src/ui/main_window_keyboard.cpp main_window_comic.cpp comic_workspace_start.cpp main_window_publication.cpp main_window_smoke.cpp comic_window_smoke.cpp catalog_window_smoke.cpp import_review_ui.cpp imported_window_smoke.cpp)
target_include_directories(inde BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/native/include ${CMAKE_CURRENT_SOURCE_DIR}/overlay/include
  ${CMAKE_CURRENT_SOURCE_DIR} ${INDE_ACTIVE_SOURCE}/src)
add_library(inde_publication_ui publication_ui.cpp publication_catalog_ui.cpp comic_ui.cpp comic_workspace_state.cpp)
target_sources(inde PRIVATE review_integration_smoke.cpp)
target_link_libraries(inde_publication_ui PUBLIC inde_web_publication PkgConfig::GTKMM)
target_link_libraries(inde PRIVATE inde_publication_ui)
set_target_properties(inde PROPERTIES OUTPUT_NAME inde-especial RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
if(INDE_USE_GTK_PCH)
  target_precompile_headers(inde PRIVATE <gtkmm.h>)
endif()
add_executable(inde_publication_tests service_tests.cpp)
target_link_libraries(inde_publication_tests PRIVATE inde_web_publication)
add_executable(inde_publication_ui_smoke ui_smoke.cpp)
target_link_libraries(inde_publication_ui_smoke PRIVATE inde_publication_ui)
add_executable(inde_catalog_tests catalog_tests.cpp)
target_link_libraries(inde_catalog_tests PRIVATE inde_web_publication)
add_executable(inde_import_tests import_tests.cpp)
target_link_libraries(inde_import_tests PRIVATE inde_web_publication)
add_executable(inde_work_import_tests work_import_tests.cpp)
target_link_libraries(inde_work_import_tests PRIVATE inde_web_publication)

add_executable(inde_comic_tests comic_tests.cpp)
target_link_libraries(inde_comic_tests PRIVATE inde_web_publication)
add_executable(inde_comic_demo comic_demo.cpp)
target_link_libraries(inde_comic_demo PRIVATE inde_web_publication PkgConfig::GTKMM)

if(INDE_BUILD_CORE_TESTS)
  get_property(INDE_BASE_TARGETS DIRECTORY ${INDE_ACTIVE_SOURCE} PROPERTY BUILDSYSTEM_TARGETS)
  foreach(candidate IN LISTS INDE_BASE_TARGETS)
    if(candidate MATCHES "_tests$")
      list(APPEND INDE_REGRESSION_TARGETS ${candidate})
    endif()
  endforeach()
  add_custom_target(inde_core_regression DEPENDS ${INDE_REGRESSION_TARGETS})
  foreach(candidate IN LISTS INDE_BASE_TARGETS)
    get_target_property(candidate_type ${candidate} TYPE)
    if(candidate_type STREQUAL "EXECUTABLE")
      list(APPEND INDE_NATIVE_EXECUTABLES ${candidate})
    endif()
  endforeach()
  add_custom_target(inde_all_native_checks DEPENDS ${INDE_NATIVE_EXECUTABLES})
endif()
if(INDE_ENABLE_LTO AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  get_property(INDE_LOCAL_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
  get_property(INDE_UPSTREAM_TARGETS DIRECTORY ${INDE_ACTIVE_SOURCE} PROPERTY BUILDSYSTEM_TARGETS)
  foreach(candidate IN LISTS INDE_LOCAL_TARGETS INDE_UPSTREAM_TARGETS)
    get_target_property(candidate_type ${candidate} TYPE)
    if(candidate_type STREQUAL "EXECUTABLE" AND NOT candidate STREQUAL "inde"
       AND NOT candidate STREQUAL "inde_web_bridge")
      target_link_options(${candidate} PRIVATE "$<$<CONFIG:Release>:-fno-lto>")
    endif()
  endforeach()
endif()
